{{/* ========================================================= BLOQUE TEMPORAL CONDICIONAL (MULTISÁBADO) --------------------------------------------------------- Muestra uno de estos partials durante los 6 días previos al sábado correspondiente (domingo → viernes anteriores): - bloque-temporada-sabado1.html → 6 días antes del 1er sábado - bloque-temporada-sabado2.html → 6 días antes del 2º sábado - bloque-temporada-sabado3.html → 6 días antes del 3er sábado - bloque-temporada-sabado4.html → 6 días antes del 4º sábado Meses activos: octubre (10) → julio (7) Excluye: agosto (8) y septiembre (9) Previsualización con hugo server: HUGO_BANNER_FECHA=2026-03-09 hugo server (o define site.Params.bannerFecha en config) Uso desde el index: {{ partial "bloque-temporada-condicional-sabados.html" . }} ========================================================= */}} {{- $zonaHoraria := "Europe/Madrid" -}} {{- /* --------------------------------------------------------- 1) Fecha "de hoy": real o simulada para previsualizar --------------------------------------------------------- */ -}} {{- $fechaSimulada := "" -}} {{- with os.Getenv "HUGO_BANNER_FECHA" -}} {{- $fechaSimulada = . -}} {{- else -}} {{- with site.Params.bannerFecha -}} {{- $fechaSimulada = . -}} {{- end -}} {{- end -}} {{- $hoy := time.Now -}} {{- if $fechaSimulada -}} {{- $hoy = time.AsTime $fechaSimulada $zonaHoraria -}} {{- end -}} {{- $hoy = $hoy | time.In $zonaHoraria -}} {{- /* Normalizamos a medianoche para comparar solo fechas */ -}} {{- $hoyTexto := $hoy | time.Format "2006-01-02" -}} {{- $hoy = time.AsTime $hoyTexto $zonaHoraria -}} {{- /* --------------------------------------------------------- 2) Mes y año de hoy --------------------------------------------------------- */ -}} {{- $mes := int ($hoy.Format "1") -}} {{- $anio := int ($hoy.Format "2006") -}} {{- /* --------------------------------------------------------- 3) ¿El mes está en rango octubre → julio? --------------------------------------------------------- */ -}} {{- $mesEnRango := or (ge $mes 10) (le $mes 7) -}} {{- $partialMostrar := "" -}} {{- if $mesEnRango -}} {{- /* ----------------------------------------------------- 4) Calcular el 1er sábado del mes actual ----------------------------------------------------- */ -}} {{- $primerDiaTexto := printf "%04d-%02d-01" $anio $mes -}} {{- $primerDia := time.AsTime $primerDiaTexto $zonaHoraria -}} {{- /* Día de la semana ISO del día 1: lunes=1 ... domingo=7. Sábado = ISO 6. */ -}} {{- $isoPrimer := 0 -}} {{- $wdPrimer := $primerDia.Weekday -}} {{- if eq $wdPrimer 0 -}} {{- $isoPrimer = 7 -}} {{- else -}} {{- $isoPrimer = int $wdPrimer -}} {{- end -}} {{- /* Días hasta el primer sábado (sábado = ISO 6) */ -}} {{- $diasHastaPrimerSabado := sub 6 $isoPrimer -}} {{- if lt $diasHastaPrimerSabado 0 -}} {{- $diasHastaPrimerSabado = add $diasHastaPrimerSabado 7 -}} {{- end -}} {{- $primerSabado := $primerDia.AddDate 0 0 $diasHastaPrimerSabado -}} {{- /* ----------------------------------------------------- 5) Recorremos los sábados 1º, 2º, 3º y 4º y comprobamos su ventana de 6 días previos ----------------------------------------------------- */ -}} {{- range $indice := seq 0 3 -}} {{- $sabado := $primerSabado.AddDate 0 0 (mul $indice 7) -}} {{- $inicioVentana := $sabado.AddDate 0 0 -6 -}} {{- $finVentana := $sabado.AddDate 0 0 -1 -}} {{/* viernes */}} {{- $enVentana := and (not ($hoy.Before $inicioVentana)) (not ($hoy.After $finVentana)) -}} {{- if $enVentana -}} {{- $num := add $indice 1 -}} {{- $partialMostrar = printf "bloque-temporada-sabado%d.html" $num -}} {{- end -}} {{- end -}} {{- end -}} {{- with $partialMostrar -}} {{ partial . $ }} {{- end -}}
