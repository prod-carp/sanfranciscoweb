{{/* ========================================================= BLOQUE TEMPORAL CONDICIONAL --------------------------------------------------------- Muestra el partial "bloque-temporada-contenido.html" durante los 6 días previos al 2º domingo del mes (lunes → sábado anteriores a ese domingo). Meses activos: octubre (10) → julio (7) Excluye: agosto (8) y septiembre (9) Previsualización con hugo server: HUGO_BANNER_FECHA=2026-03-09 hugo server (o define site.Params.bannerFecha en config) Uso desde el index: {{ partial "bloque-temporada-condicional.html" . }} ========================================================= */}} {{- $zonaHoraria := "Europe/Madrid" -}} {{- /* --------------------------------------------------------- 1) Fecha "de hoy": real o simulada para previsualizar --------------------------------------------------------- */ -}} {{- $fechaSimulada := "" -}} {{- with os.Getenv "HUGO_BANNER_FECHA" -}} {{- $fechaSimulada = . -}} {{- else -}} {{- with site.Params.bannerFecha -}} {{- $fechaSimulada = . -}} {{- end -}} {{- end -}} {{- $hoy := time.Now -}} {{- if $fechaSimulada -}} {{- $hoy = time.AsTime $fechaSimulada $zonaHoraria -}} {{- end -}} {{- $hoy = $hoy | time.In $zonaHoraria -}} {{- /* Normalizamos a medianoche para comparar solo fechas */ -}} {{- $hoyTexto := $hoy | time.Format "2006-01-02" -}} {{- $hoy = time.AsTime $hoyTexto $zonaHoraria -}} {{- /* --------------------------------------------------------- 2) Mes y año de hoy --------------------------------------------------------- */ -}} {{- $mes := int ($hoy.Format "1") -}} {{- $anio := int ($hoy.Format "2006") -}} {{- /* --------------------------------------------------------- 3) ¿El mes está en rango octubre → julio? --------------------------------------------------------- */ -}} {{- $mesEnRango := or (ge $mes 10) (le $mes 7) -}} {{- $mostrar := false -}} {{- if $mesEnRango -}} {{- /* ----------------------------------------------------- 4) Calcular el 2º domingo del mes actual ----------------------------------------------------- */ -}} {{- $primerDiaTexto := printf "%04d-%02d-01" $anio $mes -}} {{- $primerDia := time.AsTime $primerDiaTexto $zonaHoraria -}} {{- /* Día de la semana ISO del día 1: lunes=1 ... domingo=7 */ -}} {{- $isoPrimer := 0 -}} {{- $wdPrimer := $primerDia.Weekday -}} {{- if eq $wdPrimer 0 -}} {{- $isoPrimer = 7 -}} {{- else -}} {{- $isoPrimer = int $wdPrimer -}} {{- end -}} {{- /* Días hasta el primer domingo (domingo = ISO 7) */ -}} {{- $diasHastaPrimerDomingo := sub 7 $isoPrimer -}} {{- $primerDomingo := $primerDia.AddDate 0 0 $diasHastaPrimerDomingo -}} {{- /* 2º domingo */ -}} {{- $segundoDomingo := $primerDomingo.AddDate 0 0 7 -}} {{- /* ----------------------------------------------------- 5) Ventana de visualización: 6 días antes del 2º domingo (lunes anterior → sábado anterior al domingo) ----------------------------------------------------- */ -}} {{- $inicioVentana := $segundoDomingo.AddDate 0 0 -6 -}} {{- $finVentana := $segundoDomingo.AddDate 0 0 -1 -}} {{/* sábado */}} {{- /* Comparación por fecha (ya normalizadas) */ -}} {{- $mostrar = and (not ($hoy.Before $inicioVentana)) (not ($hoy.After $finVentana)) -}} {{- end -}} {{- if $mostrar -}} {{ partial "bloque-temporada-contenido.html" . }} {{- end -}}
