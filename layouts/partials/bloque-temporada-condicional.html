{{/* ========================================================= BLOQUE TEMPORAL CONDICIONAL (MULTIDOMINGO) --------------------------------------------------------- Muestra uno de estos partials durante los 6 días previos al domingo correspondiente (lunes → sábado anteriores): - bloque-temporada-domingo1.html → 6 días antes del 1er domingo - bloque-temporada-domingo2.html → 6 días antes del 2º domingo - bloque-temporada-domingo3.html → 6 días antes del 3er domingo - bloque-temporada-domingo4.html → 6 días antes del 4º domingo Al partial se le pasa un dict con: - "page" → el contexto de la página (.) - "fecha" → time.Time del domingo correspondiente - "numero" → 1, 2, 3 o 4 - "nombre" → "domingo" ========================================================= */}} {{- $zonaHoraria := "Europe/Madrid" -}} {{- $fechaSimulada := "" -}} {{- with os.Getenv "HUGO_BANNER_FECHA" -}} {{- $fechaSimulada = . -}} {{- else -}} {{- with site.Params.bannerFecha -}} {{- $fechaSimulada = . -}} {{- end -}} {{- end -}} {{- $hoy := time.Now -}} {{- if $fechaSimulada -}} {{- $hoy = time.AsTime $fechaSimulada $zonaHoraria -}} {{- end -}} {{- $hoy = $hoy | time.In $zonaHoraria -}} {{- $hoyTexto := $hoy | time.Format "2006-01-02" -}} {{- $hoy = time.AsTime $hoyTexto $zonaHoraria -}} {{- $mes := int ($hoy.Format "1") -}} {{- $anio := int ($hoy.Format "2006") -}} {{- $mesEnRango := or (ge $mes 10) (le $mes 7) -}} {{- $partialMostrar := "" -}} {{- $fechaMostrar := "" -}} {{- $numeroMostrar := 0 -}} {{- if $mesEnRango -}} {{- $primerDiaTexto := printf "%04d-%02d-01" $anio $mes -}} {{- $primerDia := time.AsTime $primerDiaTexto $zonaHoraria -}} {{- $isoPrimer := 0 -}} {{- $wdPrimer := $primerDia.Weekday -}} {{- if eq $wdPrimer 0 -}} {{- $isoPrimer = 7 -}} {{- else -}} {{- $isoPrimer = int $wdPrimer -}} {{- end -}} {{- $diasHastaPrimerDomingo := sub 7 $isoPrimer -}} {{- $primerDomingo := $primerDia.AddDate 0 0 $diasHastaPrimerDomingo -}} {{- range $indice := seq 0 3 -}} {{- $domingo := $primerDomingo.AddDate 0 0 (mul $indice 7) -}} {{- $inicioVentana := $domingo.AddDate 0 0 -6 -}} {{- $finVentana := $domingo.AddDate 0 0 -1 -}} {{- $enVentana := and (not ($hoy.Before $inicioVentana)) (not ($hoy.After $finVentana)) -}} {{- if $enVentana -}} {{- $numeroMostrar = add $indice 1 -}} {{- $partialMostrar = printf "bloque-temporada-domingo%d.html" $numeroMostrar -}} {{- $fechaMostrar = $domingo -}} {{- end -}} {{- end -}} {{- end -}} {{- with $partialMostrar -}} {{ partial . (dict "page" $ "fecha" $fechaMostrar "numero" $numeroMostrar "nombre" "domingo" ) }} {{- end -}}
