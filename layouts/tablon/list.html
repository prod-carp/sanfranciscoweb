TEMP vigentes 2. Carteles normales 3. ROT según rotación diaria ============================================================ */}} {{ $zonaHoraria := "Europe/Madrid" }} {{ $saltoRotacion := 2 }} {{ $extensiones := slice ".webp" ".jpg" ".jpeg" ".png" }} {{/* Fecha de hoy a medianoche en Madrid */}} {{ $ahora := time.Now | time.In $zonaHoraria }} {{ $hoyTexto := $ahora | time.Format "2006-01-02" }} {{ $hoy := time.AsTime $hoyTexto $zonaHoraria }} {{ $archivos := readDir "static/carteles" }} {{ $temporales := slice }} {{ $normales := slice }} {{ $rotativos := slice }} {{ range $archivo := $archivos }} {{ if not $archivo.IsDir }} {{ $nombre := $archivo.Name }} {{ $extension := path.Ext $nombre | lower }} {{ if in $extensiones $extension }} {{/* ============ TEMPORALES ============ */}} {{ if hasPrefix $nombre "TEMP-" }} {{ $partes := split $nombre "-" }} {{ if ge (len $partes) 4 }} {{ $fechaTexto := printf "%s-%s-%s" (index $partes 1) (index $partes 2) (index $partes 3) }} {{ $fechaCaducidad := time.AsTime $fechaTexto $zonaHoraria }} {{/* Visible durante todo el día de caducidad */}} {{ if not ($fechaCaducidad.Before $hoy) }} {{ $temporales = $temporales | append $archivo }} {{ end }} {{ end }} {{/* ============ ROTATIVOS ============ */}} {{ else if hasPrefix $nombre "ROT-" }} {{ $rotativos = $rotativos | append $archivo }} {{/* ============ NORMALES ============ */}} {{ else }} {{ $normales = $normales | append $archivo }} {{ end }} {{ end }} {{ end }} {{ end }} {{/* -------------------- ORDENAR -------------------- */}} {{ $temporales = sort $temporales "Name" "asc" }} {{ $normales = sort $normales "Name" "asc" }} {{ $rotativos = sort $rotativos "Name" "asc" }} {{/* -------------------- ROTACIÓN DIARIA -------------------- Mismo algoritmo determinista que el carrusel. Referencia fija: 2026-01-01 (UTC). Se avanza $saltoRotacion posiciones por día. --------------------------------------------------------- */}} {{ $rotacion := slice }} {{ if gt (len $rotativos) 0 }} {{ $fechaReferencia := time.AsTime "2026-01-01" "UTC" }} {{ $hoyUTC := time.AsTime $hoyTexto "UTC" }} {{ $duracion := $hoyUTC.Sub $fechaReferencia }} {{ $diasTranscurridos := int (div $duracion.Hours 24) }} {{ if lt $diasTranscurridos 0 }} {{ $diasTranscurridos = 0 }} {{ end }} {{ $cantidadRotativos := len $rotativos }} {{ $inicio := mod (mul $diasTranscurridos $saltoRotacion) $cantidadRotativos }} {{ range $i := seq 0 (sub $cantidadRotativos 1) }} {{ $indice := mod (add $inicio $i) $cantidadRotativos }} {{ $rotacion = $rotacion | append (index $rotativos $indice) }} {{ end }} {{ end }} {{/* -------------------- LISTA FINAL -------------------- TEMP → NORMALES → ROT (rotados) --------------------------------------------------------- */}} {{ $carteles := $temporales | append $normales | append $rotacion }} {{ range $carteles }} {{ $nombre := .Name }} {{ end }}
